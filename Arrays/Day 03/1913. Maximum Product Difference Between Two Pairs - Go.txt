func maxProductDifference(nums []int) int {
    var min1 int = math.MaxInt    
    var min2 int = math.MaxInt    
    var max1 int = math.MinInt  
    var max2 int = math.MinInt
       
    max1_index := 0
    min1_index := 0

    for i:= 0;i < len(nums) ; i++ {
        if max1 < nums[i] {
            max1=nums[i]
            max1_index = i
        }
        if min1 > nums[i] {
            min1=nums[i]
            min1_index = i
        }
    }
    for i:= 0;i < len(nums) ; i++ {
        if max2 < nums[i] && max1_index!=i {
            max2=nums[i]
        }
        if min2 > nums[i] && min1_index!=i{
            min2=nums[i]
        }
    }
    return (max1*max2 - min1*min2)
}
